## 🎲 Transition Function (Core Engine)

```sql
CREATE OR REPLACE FUNCTION simulate_ckd_progression(p_id UUID, steps INT)
RETURNS VOID AS $$
DECLARE
    t INT := 0;
    current_state TEXT := 'RISK';
    current_egfr NUMERIC := 95;

    bp NUMERIC;
    sugar NUMERIC;

    epsilon NUMERIC;
    shock BOOLEAN;

BEGIN
    -- pull baseline risk
    SELECT systolic_bp, hba1c INTO bp, sugar
    FROM risk_factors
    WHERE patient_id = p_id
    ORDER BY recorded_at DESC
    LIMIT 1;

    WHILE t < steps LOOP

        -- ε: heavy-tailed noise
        epsilon := (random() - 0.5) * 10;

        -- rare shock event (Macduff)
        shock := random() < 0.05;

        IF shock THEN
            epsilon := epsilon * 5;
        END IF;

        -- disease progression (decline in eGFR)
        current_egfr := current_egfr 
            - (0.5 + 0.01*bp + 0.02*sugar + epsilon);

        -- state transitions
        IF current_egfr > 60 THEN
            current_state := 'CKD_1_2';
        ELSIF current_egfr > 30 THEN
            current_state := 'CKD_3_4';
        ELSIF current_egfr > 15 THEN
            current_state := 'ESRD';
        ELSIF current_egfr > 10 THEN
            current_state := 'DIALYSIS';
        ELSE
            current_state := 'TRANSPLANT_OR_DEATH';
        END IF;

        INSERT INTO patient_simulation(patient_id, state, egfr, time_step)
        VALUES (p_id, current_state, current_egfr, t);

        t := t + 1;
    END LOOP;
END;
$$ LANGUAGE plpgsql;
```

function eGFRToState(egfr) {
  if (egfr > 60) return 0;
  if (egfr > 45) return 1;
  if (egfr > 30) return 2;
  if (egfr > 15) return 3;
  if (egfr > 10) return 4;
  return 5;
}

function rand(min, max) { return min + Math.random()*(max-min); }
function randN(mean, sd) {
  let u=0,v=0;
  while(u===0) u=Math.random();
  while(v===0) v=Math.random();
  return mean + sd * Math.sqrt(-2*Math.log(u)) * Math.cos(2*Math.PI*v);
}

function simulateTrajectory(baseEgfr, sbp, hba1c, eta, shockProb, steps, bpControlAt) {
  let egfr = baseEgfr;
  let traj = [egfr];
  let shocks = [];
  for (let t=1; t<=steps; t++) {
    let bpFactor = bpControlAt !== null && t >= bpControlAt ? (sbp*0.85) : sbp;
    let decline = eta * (0.3 + 0.003*bpFactor + 0.015*(hba1c-5) + randN(0,0.8));
    let shock = Math.random() < shockProb;
    if (shock) {
      decline += rand(2,8);
      shocks.push(t);
    }
    egfr = Math.max(5, egfr - decline);
    traj.push(+egfr.toFixed(1));
  }
  return { traj, shocks };
}

function rn(){let u=0,v=0;while(!u)u=Math.random();while(!v)v=Math.random();return Math.sqrt(-2*Math.log(u))*Math.cos(2*Math.PI*v);}
function rng(mn,mx){return mn+Math.random()*(mx-mn);}
function rnorm(m,s){return m+s*rn();}
function clamp(v,a,b){return Math.max(a,Math.min(b,v));}

const RACE_MULT={black:1.0,white:1.0,hispanic:0.95,asian:0.92,other:1.0};
const RACE_NAMES={black:'1.00 (ref)',white:'1.00',hispanic:'0.95',asian:'0.92',other:'1.00'};

function kfre(age,male,egfr,uacr){
  let lp=-0.2201*(age/10-7.036)+0.2467*(male-0.5642)-0.5567*(egfr/5-7.222)+0.4510*(Math.log(uacr)-5.137);
  return 1-Math.pow(0.9328,Math.exp(lp));
}

function egfrToState(e){
  if(e>60)return 0;if(e>45)return 1;if(e>30)return 2;if(e>15)return 3;if(e>10)return 4;return 5;
}

function ckdStage(e){
  if(e>=90)return'G1';if(e>=60)return'G2';if(e>=45)return'G3a';if(e>=30)return'G3b';if(e>=15)return'G4';return'G5';
}

function kdigoRisk(e,uacr){
  let eu=egfrToState(e);
  let au=uacr<30?0:uacr<300?1:2;
  let score=eu*3+au;
  if(score<=1)return'Low';if(score<=4)return'Moderate';if(score<=7)return'High';return'Very High';
}

function smkDelta(s){return s==='never'?0:s==='former'?-0.3:-0.8;}
function htnLabel(sbp){return sbp<130?'Controlled':sbp<140?'Borderline':'Uncontrolled';}

function monthlyDecline(egfr,sbp,hba1c,dm,htn,smk,race){
  let base=0.28;
  base+=0.0025*(sbp-120);
  if(dm==='1'||dm===1)base+=0.12;
  if(htn==='1'||htn===1)base+=0.08;
  base+=Math.max(0,(hba1c-5.5)*0.04);
  base+=smkDelta(smk)*(-0.05);
  base*=RACE_MULT[race]||1;
  return Math.max(0.05,base);
}

let simTraj=[], simInt=[], simShocks=[];

function simulate(egfr,sbp,hba1c,dm,htn,smk,race,eps,steps,bpTarget){
  let t=[egfr],shk=[];
  let e=egfr;
  for(let i=1;i<=steps;i++){
    let sbpEff=bpTarget!==null&&i>=3?bpTarget:sbp;
    let dec=monthlyDecline(e,sbpEff,hba1c,dm,htn,smk,race)+rnorm(0,0.4);
    if(Math.random()<eps){dec+=rng(2,7);shk.push(i);}
    e=Math.max(4,e-dec);
    t.push(+e.toFixed(1));
  }
  return{t,shk};
}

  base+=0.003*(sbpE-120);
