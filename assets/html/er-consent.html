Preface: The Failure of Static Consent

The architecture of human care is currently suffering from a loss of fidelity. Traditional "informed consent" is a lossy compression—a static paper waiver that fails to map the uncompressed dimensions of a patient's will. When a system forces an irreversible update ($ \theta' $) without calibrating against the bedrock of lived experience, the resulting divergence ($ \epsilon $) creates trauma. This stack aims to fix that by making consent a runtime gate.

from fastapi import FastAPI, HTTPException, Query
from fastapi.responses import HTMLResponse
from pydantic import BaseModel, Field
from datetime import datetime
from typing import Dict, List, Literal
from uuid import UUID, uuid4
import os

app = FastAPI(title="Ukubona Consent Architecture", version="1.0.0")

# --- SCHEMA: THE CONSENT STATE (C_t) ---
class ConsentState(BaseModel):
    id: UUID = Field(default_factory=uuid4)
    patient_id: str
    weights: Dict[str, float]  # e.g., {"dignity": 0.6, "survival": 0.4}
    reversibility: float = 1.0  # 1.0 = fully reversible, 0.0 = irreversible
    valid: bool = True

class ExternalityLedgerEntry(BaseModel):
    id: UUID = Field(default_factory=uuid4)
    action: str
    externality_score: float
    status: str = "pending"
    timestamp: datetime = Field(default_factory=datetime.utcnow)

# In-memory storage for the pilot
consent_db: Dict[str, ConsentState] = {}
ledger: List[ExternalityLedgerEntry] = []

@app.get("/", response_class=HTMLResponse)
async def root():
    with open("index.html") as f:
        return f.read()

@app.post("/consent/init")
async def init_consent(patient_id: str, weights: Dict[str, float]):
    state = ConsentState(patient_id=patient_id, weights=weights)
    consent_db[patient_id] = state
    return state

@app.post("/intervene/propose")
async def propose(patient_id: str, action: str, impact: float):
    if patient_id not in consent_db:
        raise HTTPException(404, "Consent state not found")
    
    current = consent_db[patient_id]
    # Simple Externality Tax: if impact > dignity weight and action is irreversible
    tax = (impact * (1 - current.reversibility))
    
    entry = ExternalityLedgerEntry(action=action, externality_score=tax)
    ledger.append(entry)
    
    if tax > 0.5:
        entry.status = "BLOCKED: High Externality"
        return {"status": "REJECTED", "tax": tax, "reason": "Consent gate fired"}
    
    entry.status = "APPROVED"
    return {"status": "EXECUTED", "tax": tax}

@app.get("/ledger")
async def get_ledger():
    return ledger
PYEOF
